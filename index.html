    async function startHgsDatabaseHandshake() {
        try {
            const token = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
            if (token) {
                await signInWithCustomToken(auth, token);
            } else {

    async function establishLiveSync() {
        if (!auth.currentUser) return;

        // 1. Season Config Sync
        const seasonDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'seasonConfig', 'main');
        onSnapshot(seasonDocRef, async (docSnap) => {
            if (docSnap.exists()) {
                seasonConfig = docSnap.data();
                if (document.getElementById('admin-season-start')) {
                    document.getElementById('admin-season-start').value = seasonConfig.start;
                    document.getElementById('admin-season-end').value = seasonConfig.end;
                }
                updateHomeDeadlineWidget();
            } else {
                await setDoc(seasonDocRef, seasonConfig);
            }
        }, (err) => console.warn("Season sync error", err));

        // 2. Users Database Sync
        const usersColRef = collection(db, 'artifacts', appId, 'public', 'data', 'users');
        onSnapshot(usersColRef, async (snap) => {
            const list = [];
            snap.forEach(d => list.push(d.data()));
            if (list.length > 0) {
                users = list;
            } else {
                for (const u of users) {
                    await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'users', u.id), u);
                }
            }
            
            // 새로고침 시 세션 복구 처리
            attemptSessionRestore();

            updateSessionBadge();
            if (activeTab === 'admin') renderAdminDashboard();
            if (activeTab === 'library' && currentUser) renderMyLibrary(currentUser.id);
        }, (err) => console.warn("Users sync error", err));

        // 3. Book Logs Sync
        const logsColRef = collection(db, 'artifacts', appId, 'public', 'data', 'bookLogs');
        onSnapshot(logsColRef, async (snap) => {
            const list = [];
            snap.forEach(d => {
                const data = d.data();
                // isRecommended 필드가 부재할 경우 강제 매핑 및 보정
                if (data.isRecommended === undefined) {
                    data.isRecommended = false;
                }
                list.push(data);
            });
            bookLogs = list;
            
            renderFriendsPick();
            if (activeTab === 'library' && currentUser) renderMyLibrary(currentUser.id);
            if (activeTab === 'admin') renderAdminDashboard();
            
            document.getElementById('loading-sync-overlay').classList.add('hidden');
        }, (err) => console.warn("Logs sync error", err));
    }

        const approvedAndCompletedLogs = bookLogs.filter(log => log.userId === user.id && log.approved === 'approved' && log.completed);
        
        const myTotalCount = approvedAndCompletedLogs.length;

        let totalPct = Math.round((myTotalCount / targets.total) * 100);

        const recommendedCount = bookLogs.filter(log => log.userId === user.id && log.approved === 'approved' && log.isRecommended).length;

        const recommendedLeft = Math.max(0, recommendedTarget - recommendedCount);

        let sub = "THE BEGINNER";
        let avatar = "⚡";

    // STREAMING_CHUNK: CSV spreadsheet compiler (Only 완독 counted in stats)
    function downloadExcelReport(userId) {
        const targetId = userId || (currentUser ? currentUser.id : null);
        if (!targetId) {
            showToast("⚠️ 리포트를 내려받을 유저 정보를 발견하지 못했습니다.");
            return;
        }

        const uObj = users.find(x => x.id === targetId);
        if (!uObj) return;

        const myLogs = bookLogs.filter(l => l.userId === uObj.id && l.approved === 'approved');

        let csvContent = "data:text/csv;charset=utf-8,\uFEFF";
        
        csvContent += `HGS Reading Challenge - 학생 독서 성과 종합 엑셀 보고서\n`;
        csvContent += `학생 고유 ID,${uObj.id}\n`;
        csvContent += `학생 성함,${uObj.name}\n`;
        csvContent += `학교 구분,${uObj.grade === 'high' ? '고등부' : '중등부'}\n`;
        csvContent += `참여 이벤트 목록,"${uObj.events.join(', ')}"\n\n`;

        csvContent += "도서ID,도서제목,작가명,출판사,완독여부,평가별점(1~5),진도범위(시작),진도범위(종료),전체페이지수,서평기록일자,권장도서여부,나의 성찰 코멘트\n";

        myLogs.forEach(l => {
            const sanitizedNotes = (l.notes || '').replace(/"/g, '""').replace(/\n/g, ' ');
            const recStatus = l.isRecommended ? 'Y' : 'N';
            csvContent += `${l.id},"${l.title}","${l.author}","${l.publisher}",${l.completed ? '완독' : '읽는중'},${l.stars},${l.pageStart},${l.pageEnd},${l.pageTotal},${l.date},${recStatus},"${sanitizedNotes}"\n`;
        });

        const encodedUri = encodeURI(csvContent);
        const link = document.createElement("a");
        link.setAttribute("href", encodedUri);
        link.setAttribute("download", `HGS_Report_${uObj.id}_Excel.csv`);
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);

        showToast(`📥 ${uObj.name} 학생의 엑셀(CSV) 보고서 컴파일 및 저장 완료!`);
    }

    function downloadSeasonArchiveCSV() {
        if (seasonConfig.status !== 'closed') {
            showToast("⚠️ 시즌 아카이브 백업은 '시즌 마감 철회(Closed)' 모드에서 가동 가능합니다.");
            return;
        }

        const nameVal = document.getElementById('archive-filename').value.trim() || "HGS_Reading_Log_Archive";
        let csvContent = "data:text/csv;charset=utf-8,\uFEFF";
        csvContent += "Student ID,Name,Grade,Approved Books,Events,Level\n";

        users.forEach(u => {
            const appLogs = bookLogs.filter(l => l.userId === u.id && l.approved === 'approved' && l.completed);
            const targets = u.grade === 'high' ? 40 : 60;
            const myTotalCount = appLogs.length;

            let level = "HARRY";
            if (myTotalCount >= targets && u.events.length >= 1) {
                level = "MATILDA";
            } else if (myTotalCount >= 1) {
                level = "ANNE";
            }

            csvContent += `${u.id},${u.name},${u.grade === 'high' ? 'High' : 'Middle'},${myTotalCount},"${u.events.join('|')}",${level}\n`;
        });

        const encodedUri = encodeURI(csvContent);
        const link = document.createElement("a");
        link.setAttribute("href", encodedUri);
        link.setAttribute("download", `${nameVal}.csv`);
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);

        showToast(`📥 ${nameVal}.csv 아카이브 컴파일 완료!`);
    }
